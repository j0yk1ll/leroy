cmake_minimum_required(VERSION 3.8)
project(leroy_m23s_rectification_ab LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)

add_executable(m23s_ab_publisher src/m23s_ab_publisher.cpp)
target_compile_features(m23s_ab_publisher PUBLIC cxx_std_17)
target_compile_options(m23s_ab_publisher PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(m23s_ab_publisher rclcpp sensor_msgs)
install(TARGETS m23s_ab_publisher DESTINATION lib/${PROJECT_NAME})
ament_package()
